/// 只留**硬期限**，重试已经去掉（`TRIES` 3 → 1）。
/// 期限的作用变了：不再是「多试几次撞运气」，
/// 是「挂住时 25s 内失败，而不是吃光 CI 的额度」。
fn with_deadline<T: Send + 'static>(
secs: u64,
body: impl Fn() -> T + Send + Sync + 'static,
) -> T {
const TRIES: u32 = 1;
for attempt in 1..=TRIES {
let (tx, rx) = mpsc::channel();
let body = Arc::clone(&body);
thread::spawn(move || {
let _ = tx.send(body());
});
match rx.recv_timeout(Duration::from_secs(secs)) {
Ok(v) => return v,
Err(_) => eprintln!(" pty 第 {attempt} 次超过 {secs}s 没返回"),
}
}
panic!("pty 在 {secs}s 内没返回 —— 先看 ps 有没有堆积的孤儿")
}
